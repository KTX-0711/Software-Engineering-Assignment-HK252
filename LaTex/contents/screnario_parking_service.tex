\textbf{Use-case ID} \newline \textbf{Use-case name} & U2.1 \newline \textbf{Xác nhận và ghi nhận ra vào tự động} \\ \hline
\textbf{Use-case overview} & Tự động ghi nhận lượt ra/vào bằng cách quét thẻ RFID và camera nhận diện mà không cần nhân viên thao tác.\\ \hline
\textbf{Actors} & 1. Khách gửi xe có tài khoản \newline 2. HCMUT\_DATACORE  \\ \hline
\textbf{Preconditions} &
1. Thẻ sinh viên / giảng viên còn hiệu lực.\newline
2. Biển số xe đã được đăng ký trên hệ thống. \\ \hline
\textbf{Trigger} & Đầu đọc thẻ tại trạm kiểm soát bắt được tín hiệu từ thẻ của khách. \\ \hline

\textbf{Normal flow} &
1. Khách quét thẻ sinh viên/giảng viên tại cổng.\newline
2. Hệ thống đọc mã số sinh viên/giảng viên từ thẻ.\newline
3. Hệ thống kiểm tra tài khoản trên HCMUT\_DATACORE.\newline
4. Hệ thống kích hoạt camera chụp biển số xe.\newline
5. Hệ thống kiểm tra biển số có khớp với dữ liệu đã đăng ký.\newline
6. Nếu hợp lệ, hệ thống ghi nhận lượt xe và cập nhật vị trí trên Parking Map sau 5s cảm biến ổn định. \\ \hline

\textbf{Alternative flow} &
A1: Nếu khách không đăng ký gói tháng, hệ thống tự động cộng dồn phí gửi xe.\newline
A2: Nếu khách có gói tháng còn hiệu lực, hệ thống bỏ qua bước tính phí. \\ \hline

\textbf{Post conditions} & Hệ thống ghi nhận lượt vào ra. \\ \hline

\textbf{Exception flow} &
E1: Thẻ không hợp lệ (hết hạn hoặc bị khóa), hệ thống báo lỗi và rào chắn không mở.\newline
E2: Biển số không khớp với đăng ký, hệ thống báo lỗi và rào chắn không mở. \\ \hline
\end{tabularx}

\textbf{Use-case ID} \newline \textbf{Use-case name} & U2.4 \newline \textbf{Đăng ký biển số xe} \\ \hline
\textbf{Use-case overview} & Khách hàng đăng ký biển số xe lên hệ thống và chờ nhân viên xác nhận. \\ \hline
\textbf{Actors} & 1. Khách hàng có tài khoản \newline 2. Nhân viên vận hành bãi xe \\ \hline
\textbf{Preconditions} & Khách hàng đã đăng nhập vào hệ thống. \\ \hline
\textbf{Trigger} & Khách nhấn nút "Đăng ký biển số xe". \\ \hline

\textbf{Normal flow} &
1. Khách nhập biển số xe và tải ảnh giấy tờ (cà vẹt).\newline
2. Hệ thống lưu dữ liệu với trạng thái "Chờ xác nhận".\newline
3. Hệ thống gửi thông báo tới giao diện nhân viên.\newline
4. Nhân viên kiểm tra thông tin và đối chiếu giấy tờ.\newline
5. Nhân viên nhấn "Xác nhận duyệt".\newline
6. Hệ thống cập nhật trạng thái "Đã duyệt" và liên kết biển số với tài khoản. \\ \hline

\textbf{Alternative flow} &
A1: Nhân viên từ chối duyệt do thông tin không hợp lệ.\newline
A2: Hệ thống gửi thông báo từ chối và lý do cho khách hàng. \\ \hline

\textbf{Post conditions} & Biển số được lưu vào hệ thống và liên kết với tài khoản khách. \\ \hline

\textbf{Exception flow} &
E1: Lỗi định dạng hoặc thiếu dữ liệu, hệ thống yêu cầu nhập lại.\newline
E2: Biển số đã tồn tại trong hệ thống, thao tác bị chặn. \\ \hline
\end{tabularx}
